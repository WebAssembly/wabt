(module
  (tag $e)

  ;; A return_call reuses the caller's frame, so the value stack height cached
  ;; in that frame has to be re-captured for the callee.  If it is left at the
  ;; caller's height, unwinding to a handler inside the callee truncates the
  ;; value stack below the callee's parameters.
  (func (export "try-catch_all") (result i32)
    (return_call $callee (i32.const 1) (i32.const 2) (i32.const 3)))

  (func $callee (param i32 i32 i32) (result i32)
    (try (result i32)
      (do
        (throw $e))
      (catch_all
        (i32.add (local.get 0) (local.get 2)))))

  (func (export "try_table") (result i32)
    (return_call $callee2 (i32.const 4) (i32.const 5) (i32.const 6)))

  (func $callee2 (param i32 i32 i32) (result i32)
    (block $b
      (try_table (catch_all $b)
        (throw $e)))
    (i32.add (local.get 0) (local.get 2)))
)
(;; STDOUT ;;;
try-catch_all() => i32:4
try_table() => i32:10
;;; STDOUT ;;)
